+private class FastPdfView(context: Context) : View(context) {
+    private val bgPaint = Paint().apply { color = Color.parseColor("#BDBDBD") }
+    private val pageBitmaps = mutableListOf<Bitmap>()
+    private var pfd: ParcelFileDescriptor? = null
+    private var renderer: PdfRenderer? = null
+
+    private var scale = 1f
+    private var fitScale = 1f
+    private var offsetX = 0f
+    private var offsetY = 0f
+    private val pageGap = 12f
+    private val pageMargin = 16f
+    private var onZoomChanged: ((Int) -> Unit)? = null
+
+    private val gestureDetector = GestureDetector(context, object : GestureDetector.SimpleOnGestureListener() {
+        override fun onScroll(e1: MotionEvent?, e2: MotionEvent, distanceX: Float, distanceY: Float): Boolean {
+            offsetX -= distanceX
+            offsetY -= distanceY
+            clampOffsets()
+            invalidate()
+            return true
+        }
+    })
+
+    private val scaleDetector = ScaleGestureDetector(context, object : ScaleGestureDetector.SimpleOnScaleGestureListener() {
+        override fun onScale(detector: ScaleGestureDetector): Boolean {
+            val old = scale
+            val minScale = fitScale
+            val newScale = (scale * detector.scaleFactor).coerceIn(minScale, 5f)
+            val factor = newScale / old
+            scale = newScale
+
+            offsetX = detector.focusX - (detector.focusX - offsetX) * factor
+            offsetY = detector.focusY - (detector.focusY - offsetY) * factor
+
+            clampOffsets()
+            emitZoom()
+            invalidate()
+            return true
+        }
+    })
+
+    fun setOnZoomChanged(listener: (Int) -> Unit) {
+        onZoomChanged = listener
+    }
+
+    fun openFile(file: File) {
+        closeDoc()
+        if (!file.exists()) return
+        pfd = ParcelFileDescriptor.open(file, ParcelFileDescriptor.MODE_READ_ONLY)
+        renderer = PdfRenderer(pfd!!)
+        renderBitmaps()
+        requestLayout()
+        invalidate()
+    }
+
+    fun fitToScreen() {
+        scale = fitScale
+        offsetX = 0f
+        offsetY = 0f
+        emitZoom()
+        invalidate()
+    }
+
+    fun setZoomPercent(percent: Int) {
+        val p = percent.coerceIn(50, 500)
+        scale = (p / 100f).coerceIn(fitScale, 5f)
+        clampOffsets()
+        emitZoom()
+        invalidate()
+    }
+
+    private fun renderBitmaps() {
+        pageBitmaps.forEach { it.recycle() }
+        pageBitmaps.clear()
+        val r = renderer ?: return
+        val targetWidth = 1400
+        for (i in 0 until r.pageCount) {
+            r.openPage(i).use { page ->
+                val ratio = page.height.toFloat() / page.width.toFloat()
+                val w = targetWidth
+                val h = (w * ratio).roundToInt().coerceAtMost(2400)
+                val bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
+                page.render(bmp, null, null, PdfRenderer.Page.RENDER_MODE_FOR_DISPLAY)
+                pageBitmaps.add(bmp)
+            }
+        }
+    }
+
+    override fun onSizeChanged(w: Int, h: Int, oldw: Int, oldh: Int) {
+        super.onSizeChanged(w, h, oldw, oldh)
+        if (pageBitmaps.isNotEmpty()) {
+            val baseWidth = pageBitmaps.maxOf { it.width }.toFloat() + 2f * pageMargin
+            fitScale = (w / baseWidth).coerceIn(0.5f, 5f)
+            fitToScreen()
+        }
+    }
+
+    override fun onDraw(canvas: Canvas) {
+        super.onDraw(canvas)
+        canvas.drawRect(0f, 0f, width.toFloat(), height.toFloat(), bgPaint)
+        if (pageBitmaps.isEmpty()) return
+
+        canvas.save()
+        canvas.translate(offsetX, offsetY)
+        canvas.scale(scale, scale)
+
+        var y = pageMargin
+        val maxW = pageBitmaps.maxOf { it.width }.toFloat()
+        for (bmp in pageBitmaps) {
+            val x = pageMargin + (maxW - bmp.width) / 2f
+            canvas.drawBitmap(bmp, x, y, null)
+            y += bmp.height + pageGap
+        }
+        canvas.restore()
+    }
+
+    override fun onTouchEvent(event: MotionEvent): Boolean {
+        val scaleHandled = scaleDetector.onTouchEvent(event)
+        val gestureHandled = gestureDetector.onTouchEvent(event)
+        return scaleHandled || gestureHandled || super.onTouchEvent(event)
+    }
+
+    private fun clampOffsets() {
+        if (pageBitmaps.isEmpty()) return
+        val contentW = (pageBitmaps.maxOf { it.width }.toFloat() + 2f * pageMargin) * scale
+        val contentH = (pageBitmaps.sumOf { it.height }.toFloat() +
+            ((pageBitmaps.size - 1).coerceAtLeast(0) * pageGap) + 2f * pageMargin) * scale
+
+        val maxX = ((contentW - width) / 2f).coerceAtLeast(0f)
+        val maxY = ((contentH - height) / 2f).coerceAtLeast(0f)
+
+        if (scale <= fitScale + 0.001f) {
+            offsetX = 0f
+            offsetY = 0f
+        } else {
+            offsetX = offsetX.coerceIn(-maxX, maxX)
+            offsetY = offsetY.coerceIn(-maxY, maxY)
+        }
+    }
+
+    private fun emitZoom() {
+        onZoomChanged?.invoke((scale * 100f).roundToInt().coerceIn(50, 500))
+    }
+
+    private fun closeDoc() {
+        pageBitmaps.forEach { it.recycle() }
+        pageBitmaps.clear()
+        try { renderer?.close() } catch (_: Throwable) {}
+        try { pfd?.close() } catch (_: Throwable) {}
+        renderer = null
+        pfd = null
+    }
+
+    override fun onDetachedFromWindow() {
+        closeDoc()
+        super.onDetachedFromWindow()
+    }
+}

+private fun sharePdf(context: Context, file: File) {

+private class PdfFilePrintAdapter(
+    private val file: File,
+) : PrintDocumentAdapter() {
+    override fun onLayout(
+        oldAttributes: PrintAttributes?,
+        newAttributes: PrintAttributes,
+        cancellationSignal: android.os.CancellationSignal,
+        callback: LayoutResultCallback,
+        extras: android.os.Bundle?,
+    ) {
+        if (cancellationSignal.isCanceled) {
+            callback.onLayoutCancelled()
+            return
+        }
+        callback.onLayoutFinished(
+            PrintDocumentInfo.Builder(file.name)
+                .setContentType(PrintDocumentInfo.CONTENT_TYPE_DOCUMENT)
+                .build(),
+            true,
+        )
+    }
+
+    override fun onWrite(
+        pages: Array<PageRange>,
+        destination: ParcelFileDescriptor,
+        cancellationSignal: android.os.CancellationSignal,
+        callback: WriteResultCallback,
+    ) {
+        try {
+            FileInputStream(file).use { input ->
+                FileOutputStream(destination.fileDescriptor).use { output ->
+                    val buf = ByteArray(16 * 1024)
+                    while (true) {
+                        if (cancellationSignal.isCanceled) {
+                            callback.onWriteCancelled()
+                            return
+                        }
+                        val n = input.read(buf)
+                        if (n <= 0) break
+                        output.write(buf, 0, n)
+                    }
+                    output.flush()
+                }
+            }
+            callback.onWriteFinished(arrayOf(PageRange.ALL_PAGES))
+        } catch (t: Throwable) {
+            callback.onWriteFailed(t.message)
+        }
+    }
+}
